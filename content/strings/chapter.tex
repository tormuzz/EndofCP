struct Trie {
    static const int ALPHA = 26;
    struct Node {
        int nxt[ALPHA], pass = 0, end = 0;
        Node() { fill(nxt, nxt + ALPHA, -1); }
    };
    vector<Node> t;
    Trie() { t.emplace_back(); }

    void insert(const string& s) {
        int u = 0; t[u].pass++;
        for (char c : s) {
            int i = c - 'a';
            if (t[u].nxt[i] == -1) { t[u].nxt[i] = (int)t.size(); t.emplace_back(); }
            u = t[u].nxt[i]; t[u].pass++;
        }
        t[u].end++;
    }
    bool contains(const string& s) {
        int u = 0;
        for (char c : s) {
            int i = c - 'a';
            if (t[u].nxt[i] == -1) return false;
            u = t[u].nxt[i];
        }
        return t[u].end > 0;
    }
    int countPrefix(const string& s) {     // strings having s as prefix
        int u = 0;
        for (char c : s) {
            int i = c - 'a';
            if (t[u].nxt[i] == -1) return 0;
            u = t[u].nxt[i];
        }
        return t[u].pass;
    }
};
\end{lstlisting}

\sectionHeading{KMP}
\begin{lstlisting}[language=C++]
vector<int> prefixFunction(const string& s) {
    int n = s.size();
    vector<int> pi(n);
    for (int i = 1; i < n; i++) {
        int j = pi[i - 1];
        while (j && s[i] != s[j]) j = pi[j - 1];
        if (s[i] == s[j]) j++;
        pi[i] = j;
    }
    return pi;
}

vector<int> kmp(const string& s, const string& p) {   // match positions
    vector<int> pi = prefixFunction(p), res;
    int j = 0;
    for (int i = 0; i < (int)s.size(); i++) {
        while (j && s[i] != p[j]) j = pi[j - 1];
        if (s[i] == p[j]) j++;
        if (j == (int)p.size()) { res.push_back(i - j + 1); j = pi[j - 1]; }
    }
    return res;
}

vector<int> zFunction(const string& s) {
    int n = s.size();
    vector<int> z(n);
    for (int i = 1, l = 0, r = 0; i < n; i++) {
        if (i <= r) z[i] = min(r - i + 1, z[i - l]);
        while (i + z[i] < n && s[z[i]] == s[i + z[i]]) z[i]++;
        if (i + z[i] - 1 > r) l = i, r = i + z[i] - 1;
    }
    return z;
}

\sectionHeading{Manacher}
\begin{lstlisting}[language=C++]
// odd[i]  = radius of longest odd palindrome centered at i  (length 2*odd[i]-1)
// even[i] = radius of longest even palindrome centered at i (length 2*even[i])
void manacher(const string& s, vector<int>& odd, vector<int>& even) {
    int n = s.size();
    odd.assign(n, 0); even.assign(n, 0);
    for (int i = 0, l = 0, r = -1; i < n; i++) {
        int k = (i > r) ? 1 : min(odd[l + r - i], r - i + 1);
        while (i - k >= 0 && i + k < n && s[i - k] == s[i + k]) k++;
        odd[i] = k;
        if (i + k - 1 > r) l = i - k + 1, r = i + k - 1;
    }
    for (int i = 0, l = 0, r = -1; i < n; i++) {
        int k = (i > r) ? 0 : min(even[l + r - i + 1], r - i + 1);
        while (i - k - 1 >= 0 && i + k < n && s[i - k - 1] == s[i + k]) k++;
        even[i] = k;
        if (i + k - 1 > r) l = i - k, r = i + k - 1;
    }
}
\end{lstlisting}

\begin{lstlisting}[language=C++]
int minRotation(const string& s) {          // O(n), returns start index
    int n = s.size();
    string t = s + s;
    vector<int> f(2 * n, -1);
    int k = 0;
    for (int j = 1; j < 2 * n; j++) {
        int i = f[j - k - 1];
        while (i != -1 && t[j] != t[k + i + 1]) {
            if (t[j] < t[k + i + 1]) k = j - i - 1;
            i = f[i];
        }
        if (t[j] != t[k + i + 1]) {
            if (t[j] < t[k]) k = j;
            f[j - k] = -1;
        } else f[j - k] = i + 1;
    }
    return k;
}
\end{lstlisting}

\sectionHeading{Duval (Lyndon factorization)}
\begin{lstlisting}[language=C++]
vector<string> duval(const string& s) {     // O(n)
    int n = s.size(), i = 0;
    vector<string> res;
    while (i < n) {
        int j = i + 1, k = i;
        while (j < n && s[k] <= s[j]) {
            if (s[k] < s[j]) k = i; else k++;
            j++;
        }
        while (i <= k) { res.push_back(s.substr(i, j - k)); i += j - k; }
    }
    return res;
}

\sectionHeading{Aho--Corasick}
\begin{lstlisting}[language=C++]
struct AhoCorasick {
    static const int ALPHA = 26;
    vector<array<int, ALPHA>> nxt;
    vector<int> link, outLink;
    vector<vector<int>> out;

    AhoCorasick() { addNode(); }
    int addNode() {
        nxt.push_back({}); nxt.back().fill(0);
        link.push_back(0); outLink.push_back(0); out.push_back({});
        return (int)nxt.size() - 1;
    }
    void insert(const string& s, int id) {
        int u = 0;
        for (char c : s) {
            int x = c - 'a';
            if (!nxt[u][x]) nxt[u][x] = addNode();
            u = nxt[u][x];
        }
        out[u].push_back(id);
    }
    void build() {
        queue<int> q; q.push(0);
        while (!q.empty()) {
            int u = q.front(); q.pop();
            for (int c = 0; c < ALPHA; c++) {
                int v = nxt[u][c];
                if (!v) nxt[u][c] = u ? nxt[link[u]][c] : 0;
                else {
                    link[v] = u ? nxt[link[u]][c] : 0;
                    outLink[v] = out[link[v]].empty() ? outLink[link[v]] : link[v];
                    q.push(v);
                }
            }
        }
    }
};
\end{lstlisting}

\subHeading{Rabinkarp templates}

\begin{lstlisting}[language=C++]
const int MOD1 = 1000000007;
const int MOD2 = 1000000009;
const int BASE1 = 911382323;
const int BASE2 = 972663749;
vector<int> power1(MAXN), power2(MAXN);

// Precompute powers once
void precomputePowers() {
    power1[0] = power2[0] = 1;
    for (int i = 1; i < MAXN; i++) {
        power1[i] = (1LL * power1[i - 1] * BASE1) % MOD1;
        power2[i] = (1LL * power2[i - 1] * BASE2) % MOD2;
    }
}

// Rolling hash struct (uses precomputed powers)
struct RabinKarpHash {
    vector<int> hash1, hash2, Rash1, Rash2;
    int n;

    RabinKarpHash(const string &s) {
        n = (int)s.size();
        hash1.assign(n + 1, 0); hash2.assign(n + 1, 0);
        Rash1.assign(n + 1, 0); Rash2.assign(n + 1, 0);

        // forward hash
        for (int i = 0; i < n; i++) {
            hash1[i + 1] = ( (1LL * hash1[i] * BASE1) + s[i] ) % MOD1;
            hash2[i + 1] = ( (1LL * hash2[i] * BASE2) + s[i] ) % MOD2;
        }

        // reverse hash
        for (int i = 0; i < n; i++) {
            Rash1[i + 1] = ( (1LL * Rash1[i] * BASE1) + s[n - i - 1] ) % MOD1;
            Rash2[i + 1] = ( (1LL * Rash2[i] * BASE2) + s[n - i - 1] ) % MOD2;
        }
    }

    inline pair<int,int> getSubHash(int l, int r) const {
        int x1 = (hash1[r + 1] - 1LL * hash1[l] * power1[r - l + 1]) % MOD1;
        int x2 = (hash2[r + 1] - 1LL * hash2[l] * power2[r - l + 1]) % MOD2;
        if (x1 < 0) x1 += MOD1; if (x2 < 0) x2 += MOD2;
        return {x1, x2};
    }

    inline pair<int,int> getRevHash(int l, int r) const {
        int L = n - 1 - r; int R = n - 1 - l;
        int x1 = (Rash1[R + 1] - 1LL * Rash1[L] * power1[R - L + 1]) % MOD1;
        int x2 = (Rash2[R + 1] - 1LL * Rash2[L] * power2[R - L + 1]) % MOD2;
        if (x1 < 0) x1 += MOD1; if (x2 < 0) x2 += MOD2;
        return {x1, x2};
    }

    inline pair<int,int> contribution(int i,int R,char c) const {
        int val = c - 'a' + 1;
        int powerPos = R - i;
        int x1 = (1LL * val * power1[powerPos]) % MOD1;
        int x2 = (1LL * val * power2[powerPos]) % MOD2;
        return {(int)x1, (int)x2};
    }

    //does not affect the main string // O(1)
    inline pair<int,int> CNGHash(int i, char oldc, char newc) const {
        auto ex = contribution(i,n-1,oldc);
        auto cur = contribution(i,n-1,newc);
        pair<int,int> p = {hash1[n], hash2[n]};
        p.first = (p.first - ex.first + cur.first + MOD1) % MOD1;
        p.second = (p.second - ex.second + cur.second + MOD2) % MOD2;
        return p;
    }
};
\end{lstlisting}

vector<int> occurrences;

void searchPattern(const string &text, const string &pattern) {
    occurrences.clear();
    int n = sz(text), m = sz(pattern);
    if (m > n) return;

    RabinKarpHash txtHash(text);    // in main function
    RabinKarpHash patHash(pattern);
    pair<int,int> patH = patHash.getSubHash(0, m - 1);

    for (int i = 0; i + m - 1 < n; i++) {
        if (txtHash.getSubHash(i, i + m - 1) == patH)
            occurrences.push_back(i);
    }
    // store 0-based index
}

\begin{lstlisting}[language=C++]
pair<int,int> hashAfterRemove(const RabinKarpHash &H, int k, int len) {
    int n = H.n;
    if (len <= 0) return n ? H.getSubHash(0, n-1) : pair<int,int>{0,0};
    if (k < 0 || k >= n || k + len > n) return {0,0};
    if (len >= n) return {0,0};

    int left_len  = k;
    int right_len = n - (k + len);

    pair<int,int> HL = left_len ?
        H.getSubHash(0, k-1) : pair<int,int>{0,0};
    pair<int,int> HR = right_len ?
        H.getSubHash(k+len, n-1) : pair<int,int>{0,0};

    int r1 = ( ( (HL.first  * power1[right_len]) % MOD1 ) + HR.first ) % MOD1;
    int r2 = ( ( (HL.second * power2[right_len]) % MOD2 ) + HR.second) % MOD2;
    return {r1, r2};
}
\end{lstlisting}
