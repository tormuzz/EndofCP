\sectionHeading{Segment tree (point update, range query)}
\begin{lstlisting}[language=C++]
struct SegTree {
    int n;
    vector<long long> t;
    SegTree(int n) : n(n), t(2 * n, 0) {}

    void build(const vector<int>& a) {          // a[0..n)
        for (int i = 0; i < n; i++) t[n + i] = a[i];
        for (int i = n - 1; i > 0; i--) t[i] = t[i << 1] + t[i << 1 | 1];
    }
    void update(int p, long long v) {           // a[p] = v
        for (t[p += n] = v; p > 1; p >>= 1)
            t[p >> 1] = t[p] + t[p ^ 1];
    }
    long long query(int l, int r) {             // sum on [l, r)
        long long res = 0;
        for (l += n, r += n; l < r; l >>= 1, r >>= 1) {
            if (l & 1) res += t[l++];
            if (r & 1) res += t[--r];
        }
        return res;
    }
};
\end{lstlisting}

\sectionHeading{Segment tree (lazy propagation: range add, range sum)}
\begin{lstlisting}[language=C++]
struct LazySegTree {
    int n;
    vector<long long> t, lazy;
    LazySegTree(int n) : n(n), t(4 * n, 0), lazy(4 * n, 0) {}

    void build(int v, int l, int r, const vector<int>& a) {
        if (l == r) { t[v] = a[l]; return; }
        int m = (l + r) >> 1;
        build(2 * v, l, m, a);
        build(2 * v + 1, m + 1, r, a);
        t[v] = t[2 * v] + t[2 * v + 1];
    }
    void apply(int v, int l, int r, long long x) {
        t[v] += x * (r - l + 1);
        lazy[v] += x;
    }
    void push(int v, int l, int r) {
        if (!lazy[v]) return;
        int m = (l + r) >> 1;
        apply(2 * v, l, m, lazy[v]);
        apply(2 * v + 1, m + 1, r, lazy[v]);
        lazy[v] = 0;
    }
    void update(int v, int l, int r, int ql, int qr, long long x) {
        if (qr < l || r < ql) return;
        if (ql <= l && r <= qr) { apply(v, l, r, x); return; }
        push(v, l, r);
        int m = (l + r) >> 1;
        update(2 * v, l, m, ql, qr, x);
        update(2 * v + 1, m + 1, r, ql, qr, x);
        t[v] = t[2 * v] + t[2 * v + 1];
    }
    long long query(int v, int l, int r, int ql, int qr) {
        if (qr < l || r < ql) return 0;
        if (ql <= l && r <= qr) return t[v];
        push(v, l, r);
        int m = (l + r) >> 1;
        return query(2 * v, l, m, ql, qr) + query(2 * v + 1, m + 1, r, ql, qr);
    }
};
\end{lstlisting}

\sectionHeading{MO's algorithm}
\begin{lstlisting}[language=C++]
struct tup{
    int block, r, l, i;
};
int n, q, distinct;
vector<tup>queries;
vector<int>cs, ans, freq;
void add (int pos){
    int x = cs[pos];
    if (freq[x] == 0) ++distinct;
    ++freq[x];
}
void remove (int pos){
    int x = cs[pos];
    --freq[x];
    if (freq[x] == 0) --distinct;
}
void solve(){
    cin >> n >> q;
    queries.resize(q); cs.resize(n); ans.resize(q); freq.resize(n);
    for (int i = 0; i < n; ++i) cin>>cs[i];
 
    vector <int>coo = cs;
    sort(coo.begin(), coo.end());
    coo.erase(unique(coo.begin(), coo.end()), coo.end());

    for(int i=0; i<n; i++)
        cs[i] = lower_bound(coo.begin(), coo.end(), cs[i]) - coo.begin();
 
    for(int i=0; i<q; i++){
        int l, r; cin>>l>>r;
        --l, --r;
        int block = l/sqrt(n);
        queries[i] = {block, (block & 1) ? -r : r, l, i};
    }
 
    sort(all(queries), [](tup &a, tup &b){
        if(a.block != b.block) return a.block < b.block;
        if(a.r != b.r) return a.r < b.r;
        return a.l < b.l;
    });

    int L=0, R=-1;
    for(int it=0; it<q; it++){
        auto [block, r, l, i] = queries[it];
        r = abs(r);
        while(R < r) add(++R);
        while(L > l) add(--L);
        while(L < l) remove(L++);
        while(R > r) remove(R--);
 
        ans[i] = distinct;
    }
 
    for(int i=0; i<q; i++) cout<<ans[i]<<nl;
}

\end{lstlisting}
