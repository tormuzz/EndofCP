\sectionHeading{Segment tree (point update, range query)}
\begin{lstlisting}[language=C++]
struct SegTree {
    int n;
    vector<long long> t;
    SegTree(int n) : n(n), t(2 * n, 0) {}

    void build(const vector<int>& a) {          // a[0..n)
        for (int i = 0; i < n; i++) t[n + i] = a[i];
        for (int i = n - 1; i > 0; i--) t[i] = t[i << 1] + t[i << 1 | 1];
    }
    void update(int p, long long v) {           // a[p] = v
        for (t[p += n] = v; p > 1; p >>= 1)
            t[p >> 1] = t[p] + t[p ^ 1];
    }
    long long query(int l, int r) {             // sum on [l, r)
        long long res = 0;
        for (l += n, r += n; l < r; l >>= 1, r >>= 1) {
            if (l & 1) res += t[l++];
            if (r & 1) res += t[--r];
        }
        return res;
    }
};
\end{lstlisting}

\sectionHeading{Segment tree (lazy propagation: range add, range sum)}
\begin{lstlisting}[language=C++]
struct LazySegTree {
    int n;
    vector<long long> t, lazy;
    LazySegTree(int n) : n(n), t(4 * n, 0), lazy(4 * n, 0) {}

    void build(int v, int l, int r, const vector<int>& a) {
        if (l == r) { t[v] = a[l]; return; }
        int m = (l + r) >> 1;
        build(2 * v, l, m, a);
        build(2 * v + 1, m + 1, r, a);
        t[v] = t[2 * v] + t[2 * v + 1];
    }
    void apply(int v, int l, int r, long long x) {
        t[v] += x * (r - l + 1);
        lazy[v] += x;
    }
    void push(int v, int l, int r) {
        if (!lazy[v]) return;
        int m = (l + r) >> 1;
        apply(2 * v, l, m, lazy[v]);
        apply(2 * v + 1, m + 1, r, lazy[v]);
        lazy[v] = 0;
    }
    void update(int v, int l, int r, int ql, int qr, long long x) {
        if (qr < l || r < ql) return;
        if (ql <= l && r <= qr) { apply(v, l, r, x); return; }
        push(v, l, r);
        int m = (l + r) >> 1;
        update(2 * v, l, m, ql, qr, x);
        update(2 * v + 1, m + 1, r, ql, qr, x);
        t[v] = t[2 * v] + t[2 * v + 1];
    }
    long long query(int v, int l, int r, int ql, int qr) {
        if (qr < l || r < ql) return 0;
        if (ql <= l && r <= qr) return t[v];
        push(v, l, r);
        int m = (l + r) >> 1;
        return query(2 * v, l, m, ql, qr) + query(2 * v + 1, m + 1, r, ql, qr);
    }
};
\end{lstlisting}

\sectionHeading{MO's algorithm}
\begin{lstlisting}[language=C++]
struct Query { int l, r, idx; };
vector<int> mo(vector<Query> q, vector<int>& a, int n) {
    int B = max(1, (int)sqrt(n));
    sort(q.begin(), q.end(), [&](const Query& x, const Query& y) {
        int bx = x.l / B, by = y.l / B;
        if (bx != by) return bx < by;
        return (bx & 1) ? x.r > y.r : x.r < y.r;   // odd-even trick
    });
    vector<int> ans(q.size());
    int curL = 0, curR = -1;
    long long cur = 0;
    auto add = [&](int i) { cur += a[i]; };       // change per problem
    auto rem = [&](int i) { cur -= a[i]; };
    for (auto& qu : q) {
        while (curL > qu.l) add(--curL);
        while (curR < qu.r) add(++curR);
        while (curL < qu.l) rem(curL++);
        while (curR > qu.r) rem(curR--);
        ans[qu.idx] = cur;
    }
    return ans;
}
\end{lstlisting}

\sectionHeading{Longest increasing subsequence}
\begin{lstlisting}[language=C++]
int lis(const vector<int>& a) {          // length, O(n log n)
    vector<int> d;
    for (int x : a) {
        auto it = lower_bound(d.begin(), d.end(), x);
        if (it == d.end()) d.push_back(x);
        else *it = x;
    }
    return (int)d.size();
}
\end{lstlisting}
