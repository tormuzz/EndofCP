void polar_sort(vector<pt> &v) { // counter-clockwise
    sort(v.begin(), v.end(), [](pt a,pt b) {
        return make_tuple(half(a), 0.0, a.norm2()) < make_tuple(half(b), (a^b), b.norm2());
    });
}

bool direction(pt a, pt b, pt c){
    return (fabsl((b-a)^(c-b))<eps && ((b-a)*(c-b))>0);
}

bool OnSegment(pt p,pt a,pt b) {
    return dcmp((a-p)^(b-p)) == 0 && dcmp((a-p)*(b-p)) < 0;
}
// determine if lines from a to b and c to d are parallel
bool LinesParallel(pt a, pt b, pt c, pt d) {
    return fabs((b-a)^(c-d)) < eps;
}
// determine if lines from a to b and c to d are collinear / same line
bool LinesCollinear(pt a, pt b, pt c, pt d) {
    return LinesParallel(a, b, c, d) &&
    fabs(orient(a, b, c)) < eps && fabs(orient(c, d, a)) < eps;
}
bool SegmentsIntersect(pt a, pt b, pt c, pt d) {
    if (LinesCollinear(a, b, c, d)){ 
        // common point
        if(dis_square(a, c)<eps || dis_square(a, d)<eps ||
            dis_square(b, c)<eps || dis_square(b, d)<eps) return 1;
        // overlapping
        if((c-a)*(c-b)>0 && (d-a)*(d-b)>0 && (c-b)*(d-b)>0) return 0;
        return 1;
    }
    // intersect
    if(orient(a, d, b)*orient(a, c, b)>0) return 0;
    if(orient(c, a, d)*orient(c, b, d)>0) return 0;
    return 1;
}

\subHeading{Convex Hull}
\begin{lstlisting}[language=C++]
vector<pt> convexHull(vector<pt>&p, bool removeRedundant) {
    if(sz(p)<=2) return 0;
    int check = removeRedundant ? 0 : -1;
    sort(p.begin(), p.end(), cmp);
    // p.erase(unique(p.begin(), p.end()), p.end());
 
    int n = p.size();
    vector<pt> ch(n+n);
    int m = 0;      // preparing lower hull
    for(int i = 0; i < n; i++) {
        while(m>1 && dcmp((ch[m-1]-ch[m-2])^(p[i]-ch[m-1])) <= check) m--;
        ch[m++] = p[i];
    }
    int k = m;      // preparing upper hull
    for(int i = n - 2; i >= 0; i--) {
        while(m>k && dcmp((ch[m-1]-ch[m-2])^(p[i]-ch[m-2])) <= check) m--;
        ch[m++] = p[i];
    }
    if(n > 1) m--;
    ch.resize(m);
    return ch;
}
\end{lstlisting}

ld polygonArea(const vector<Point>& p) {       // signed shoelace
    ld a = 0;
    for (int i = 0, n = p.size(); i < n; i++) a += p[i].cross(p[(i + 1) % n]);
    return a / 2;
}

\subHeading{Pick's theorem}
\begin{lstlisting}[language=C++]
// Pick: A = I + B/2 - 1  =>  I = A - B/2 + 1   (integer coordinates)
int boundaryPoints(const vector<pt>&p) {
    int b = 0;
    for (int i = 0, n = p.size(); i < n; i++) {
        int dx = llabs(llroundl(p[(i + 1) % n].x - p[i].x));
        int dy = llabs(llroundl(p[(i + 1) % n].y - p[i].y));
        b += __gcd(dx, dy);
    }
    return b;
}

\begin{lstlisting}[language=C++]
ld common_area_of_2_cir(pt a, ld r1, pt b, ld r2){
    ld ds=dis(a, b);
    if(ds-r1-r2>=0) return 0;
    else if(ds-fabsl(r1-r2)<=eps) return PI*min(r1, r2)*min(r1, r2);
    else{
        ld a = acos((ds*ds + r1*r1 - r2*r2)/(2*ds*r1));
        ld b = acos((ds*ds + r2*r2 - r1*r1)/(2*ds*r2));
        ld x = 0.5*sqrt((-ds+r1+r2)*(ds+r1-r2)*(ds-r1+r2)*(ds+r1+r2));
        return r1*r1*a + r2*r2*b -x;
    }
}
